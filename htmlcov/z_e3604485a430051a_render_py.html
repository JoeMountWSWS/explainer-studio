33def render_scenes(ep: Path, timing: dict, quality: str, only: set[str] | None, force: bool) -> list[Path]: 

34 cache = ep / "build" / "scenes" 

35 cache.mkdir(parents=True, exist_ok=True) 

36 assets_key = tree_hash(ep / "assets") 

37 outputs = [] 

38 for s in timing["scenes"]: 

39 scene_file = ep / "scenes" / f"{s['id']}.html" 

40 key = sha(scene_file, assets_key, s["duration"], timing["fps"], timing["width"], timing["height"], 

41 quality, HF_VERSION)[:16] 

42 target = cache / f"{s['id']}.{quality}.{key}.mp4" 

43 outputs.append(target) 

44 # Cached scenes are reused unless --force names them; missing ones always render. 

45 if target.exists() and not (force and (not only or s["id"] in only)): 

46 continue 

47 for old in cache.glob(f"{s['id']}.{quality}.*.mp4"): 

48 old.unlink() 

49 info(f" render {s['id']} ({s['duration']:.1f}s)") 

50 shell = render_shell(ep, s, timing) 

51 try: 

52 run(hf("render", "-c", shell.name, "-f", str(timing["fps"]), "-q", quality, 

53 "--quiet", "-o", str(target)), cwd=ep) 

54 finally: 

55 shell.unlink(missing_ok=True) 

56 got = ffprobe_duration(target) 

57 if abs(got - s["duration"]) > 1.5 / timing["fps"]: 

58 die(f"scene {s['id']} rendered {got:.3f}s, expected {s['duration']:.3f}s") 

59 return outputs

70def build_vo_track(ep: Path, timing: dict, dst: Path) -> None: 

71 total = int(round(timing["duration"] * SAMPLE_RATE)) 

72 track = np.zeros(total, dtype=np.float32) 

73 for s in timing["scenes"]: 

74 wav = ep / "audio" / "vo" / f"{s['id']}.wav" 

75 if not wav.exists(): 

76 continue 

77 data, _ = sf.read(str(wav), dtype="float32") 

78 at = int(round((s["start"] + s["voStart"]) * SAMPLE_RATE)) 

79 end = min(at + len(data), total) 

80 track[at:end] += data[: end - at] 

81 sf.write(str(dst), track, SAMPLE_RATE, subtype="PCM_24")

84def mix(ep: Path, script: Script, timing: dict, vo: Path, dst: Path) -> dict: 

85 """Narration (+ optional ducked music bed), mastered to the loudness target in two passes.""" 

86 dur = timing["duration"] 

87 target_i, target_tp = float(script.setting("loudness")), float(script.setting("true_peak")) 

88 music = next(iter(sorted((ep / "assets").glob("music.*"))), None) if script.setting("music") else None 

89 

90 inputs = ["-i", str(vo)] 

91 if music: 

92 inputs += ["-stream_loop", "-1", "-i", str(music)] 

93 gain = float(script.setting("music_db")) 

94 fade_in = float(script.setting("music_fade_in")) 

95 fade = f"afade=t=in:d={fade_in}," if fade_in > 0 else "" 

96 graph = ( 

97 "[0:a]aformat=channel_layouts=mono,asplit=2[v1][vkey];" 

98 f"[1:a]aformat=channel_layouts=stereo,atrim=0:{dur:.3f},volume={gain}dB," 

99 f"{fade}afade=t=out:st={max(dur - 4, 0):.3f}:d=4[m];" 

100 "[m][vkey]sidechaincompress=threshold=0.03:ratio=6:attack=40:release=600[duck];" 

101 "[v1]pan=stereo|c0=c0|c1=c0[vs];" 

102 "[vs][duck]amix=inputs=2:duration=first:normalize=0[mix]" 

103 ) 

104 else: 

105 graph = "[0:a]pan=stereo|c0=c0|c1=c0[mix]" 

106 pre = dst.with_name("premix.wav") 

107 run(["ffmpeg", "-v", "error", "-y", *inputs, "-filter_complex", graph, "-map", "[mix]", 

108 "-t", f"{dur:.3f}", "-ar", str(SAMPLE_RATE), "-c:a", "pcm_s24le", str(pre)]) 

109 

110 ln = f"loudnorm=I={target_i}:TP={target_tp}:LRA=11" 

111 # Pass 1 measures (loudnorm prints its JSON on stderr); pass 2 applies a linear gain. 

112 proc = subprocess.run(["ffmpeg", "-hide_banner", "-nostats", "-i", str(pre), "-af", 

113 f"{ln}:print_format=json", "-f", "null", "-"], capture_output=True, text=True) 

114 found = re.search(r"\{[^{}]*\"input_i\"[^{}]*\}", proc.stderr, re.S) 

115 if not found: 

116 die("loudness measurement failed:\n" + "\n".join(proc.stderr.strip().splitlines()[-25:])) 

117 m = json.loads(found.group(0)) 

118 second = (f"{ln}:measured_I={m['input_i']}:measured_TP={m['input_tp']}:measured_LRA={m['input_lra']}" 

119 f":measured_thresh={m['input_thresh']}:offset={m['target_offset']}:linear=true") 

120 run(["ffmpeg", "-v", "error", "-y", "-i", str(pre), "-af", second, "-ar", str(SAMPLE_RATE), 

121 "-c:a", "pcm_s24le", str(dst)]) 

122 pre.unlink() 

123 return m

128def _ts(t: float, sep: str = ",") -> str: 

129 ms = int(round(t * 1000)) 

130 h, ms = divmod(ms, 3_600_000) 

131 m, ms = divmod(ms, 60_000) 

132 s, ms = divmod(ms, 1000) 

133 return f"{h:02d}:{m:02d}:{s:02d}{sep}{ms:03d}"

136def _caption_cues(ep: Path, timing: dict, max_chars: int, max_secs: float, max_words: int = 99, 

137 min_sentence: int = 12) -> list[list[dict]]: 

138 """Group the script's own words (correct spelling) into cues; each word carries absolute times.""" 

139 cues = [] 

140 for s in timing["scenes"]: 

141 wf = ep / "audio" / "vo" / f"{s['id']}.words.json" 

142 words = json.loads(wf.read_text(encoding="utf-8"))["words"] if wf.exists() else [] 

143 offset = s["start"] + s["voStart"] 

144 group: list[dict] = [] 

145 

146 def flush(): 

147 if group: 

148 cues.append(list(group)) 

149 group.clear() 

150 

151 for w in words: 

152 w = {"text": w["text"], "start": offset + w["start"], "end": offset + w["end"]} 

153 text_len = len(" ".join(x["text"] for x in group + [w])) 

154 if group and (text_len > 2 * max_chars or len(group) >= max_words 

155 or w["end"] - group[0]["start"] > max_secs): 

156 flush() 

157 group.append(w) 

158 sofar = len(" ".join(x["text"] for x in group)) 

159 if re.search(r"[.!?]['\")\]]*$", w["text"]) and sofar > min_sentence: 

160 flush() # end of sentence 

161 elif re.search(r"[,;:—]['\")\]]*$", w["text"]) and sofar > max_chars: 

162 flush() # a clause break once the cue is already a full line 

163 flush() 

164 return cues

167def captions(ep: Path, timing: dict, dst: Path, max_chars: int = 42, max_secs: float = 6.0) -> int: 

168 """SRT from the script's own words (correct spelling) with aligned timings; 2 lines max.""" 

169 cues = [(g[0]["start"], g[-1]["end"], [w["text"] for w in g]) for g in _caption_cues(ep, timing, max_chars, max_secs)] 

170 lines = [] 

171 for i, (a, b, words) in enumerate(cues, 1): 

172 nxt = cues[i][0] if i < len(cues) else b + 1.0 

173 b = min(max(b + 0.25, a + 1.0), nxt - 0.04) # linger a little, never overlap the next 

174 text = " ".join(words) 

175 if len(text) > max_chars: # break into two balanced lines at a word boundary 

176 mid, best = len(text) / 2, None 

177 for k in range(1, len(words)): 

178 cut = len(" ".join(words[:k])) 

179 if best is None or abs(cut - mid) < abs(best[0] - mid): 

180 best = (cut, k) 

181 text = " ".join(words[: best[1]]) + "\n" + " ".join(words[best[1]:]) 

182 lines.append(f"{i}\n{_ts(a)} --> {_ts(b)}\n{text}\n") 

183 dst.write_text("\n".join(lines), encoding="utf-8") 

184 return len(cues)

187def chapters(timing: dict) -> tuple[list[str], list[str]]: 

188 """YouTube chapter lines (consecutive scenes sharing a title merge) and any rule violations.""" 

189 merged: list[list] = [] 

190 for s in timing["scenes"]: 

191 if merged and merged[-1][1] == s["title"]: 

192 merged[-1][2] += s["duration"] 

193 else: 

194 merged.append([s["start"], s["title"], s["duration"]]) 

195 out, problems = [], [] 

196 for start, title, dur in merged: 

197 t = int(start) 

198 stamp = f"{t // 3600}:{t % 3600 // 60:02d}:{t % 60:02d}" if t >= 3600 else f"{t // 60}:{t % 60:02d}" 

199 out.append(f"{stamp} {title}") 

200 if dur < 10: 

201 problems.append(f"chapter '{title}' is {dur:.1f}s (YouTube needs >= 10s)") 

202 if len(merged) < 3: 

203 problems.append("YouTube needs at least 3 chapters to show them") 

204 return out, problems

219def ass_captions(ep: Path, timing: dict, dst: Path) -> int: 

220 """Burn-in captions in house style: white sans on a dark tag box, the spoken word in red. 

221 Short single-line cues (a few words), one event per word so the highlight follows the voice. 

222 libass can't pick weights from the variable font, so this relies on a static bold: DINPro-Bold from 

223 brand/fonts-licensed/ when present, else brand/fonts/SourceSans3-Bold.ttf (see caption_font()).""" 

224 w, h = timing["width"], timing["height"] 

225 portrait = h > w 

226 size, y = (72, int(h * 0.69)) if portrait else (52, int(h * 0.86)) # reel caption band / lower third 

227 ink, red, white = _ass_color("#1b2129"), _ass_color("#e71e25"), _ass_color("#ffffff") 

228 font = caption_font()[0] 

229 head = [ 

230 "[Script Info]", "ScriptType: v4.00+", f"PlayResX: {w}", f"PlayResY: {h}", "WrapStyle: 2", 

231 "ScaledBorderAndShadow: yes", "", 

232 "[V4+ Styles]", 

233 "Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, Italic, " 

234 "Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, " 

235 "MarginL, MarginR, MarginV, Encoding", 

236 f"Style: Cap,{font},{size},{white},{white},{ink},{ink},-1,0,0,0,100,100,0,0,3,14,0,5,0,0,0,1", 

237 "", "[Events]", "Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text", 

238 ] 

239 cues = _caption_cues(ep, timing, max_chars=13 if portrait else 21, max_secs=2.5, max_words=4, min_sentence=0) 

240 events = [] 

241 for i, cue in enumerate(cues): 

242 nxt = cues[i + 1][0]["start"] if i + 1 < len(cues) else cue[-1]["end"] + 1.0 

243 cue_end = min(cue[-1]["end"] + 0.3, nxt) 

244 for k, word in enumerate(cue): 

245 a = cue[0]["start"] if k == 0 else word["start"] 

246 b = cue[k + 1]["start"] if k + 1 < len(cue) else cue_end 

247 text = " ".join(("{\\c" + red + "&}" + x["text"] + "{\\c" + white + "&}") if j == k else x["text"] 

248 for j, x in enumerate(cue)) 

249 if font == "DINPro-Bold": # its fi/fl ligatures are broken; a zero-width non-joiner stops HarfBuzz forming them 

250 text = re.sub(r"f(?=[il])", "f‌", text) 

251 events.append(f"Dialogue: 0,{_ts(a, '.')[1:-1]},{_ts(b, '.')[1:-1]},Cap,,0,0,0,,{{\\pos({w // 2},{y})}}{text}") 

252 dst.write_text("\n".join(head + events) + "\n", encoding="utf-8") 

253 return len(cues)

258def render_episode(ep: Path, script: Script, timing: dict, draft: bool, only: set[str] | None, force: bool) -> Path: 

259 quality = "draft" if draft else "high" 

260 build = ep / "build" 

261 build.mkdir(exist_ok=True) 

262 parts = render_scenes(ep, timing, quality, only, force) 

263 

264 info(" concatenating scenes") 

265 video = build / f"video.{quality}.mp4" 

266 concat(parts, video) 

267 

268 info(" mixing and mastering audio") 

269 vo = build / "vo.wav" 

270 build_vo_track(ep, timing, vo) 

271 master = build / "master.wav" 

272 m = mix(ep, script, timing, vo, master) 

273 

274 OUT.mkdir(exist_ok=True) 

275 name = ep.name + ("-draft" if draft else "") 

276 final = OUT / f"{name}.mp4" 

277 vcodec = ["-c:v", "copy"] 

278 if script.setting("burn_captions"): 

279 info(" burning in captions") 

280 ass = build / "captions.ass" 

281 ass_captions(ep, timing, ass) 

282 crf = "26" if draft else "16" 

283 # A relative fontsdir (ffmpeg runs in build/): the colon in a Windows drive path splits filter options. 

284 fonts = build / "fonts" 

285 shutil.copytree(caption_font()[1], fonts, dirs_exist_ok=True) 

286 vcodec = ["-vf", f"ass={ass.name}:fontsdir={fonts.name}", "-c:v", "libx264", "-preset", "medium", 

287 "-crf", crf, "-pix_fmt", "yuv420p"] 

288 run(["ffmpeg", "-v", "error", "-y", "-i", str(video.resolve()), "-i", str(master.resolve()), "-map", "0:v", "-map", "1:a", 

289 *vcodec, "-c:a", "aac", "-b:a", "384k", "-ar", str(SAMPLE_RATE), 

290 "-movflags", "+faststart", str(final.resolve())], cwd=build) 

291 

292 n = captions(ep, timing, OUT / f"{ep.name}.srt") 

293 chap, problems = chapters(timing) 

294 if timing.get("format") == "reel": 

295 problems = [] # chapters are a YouTube long-form feature 

296 (OUT / f"{ep.name}.chapters.txt").write_text("\n".join(chap) + "\n", encoding="utf-8") 

297 

298 info(f"\n {final.relative_to(OUT.parent)} {ffprobe_duration(final):.1f}s") 

299 info(f" loudness in {m['input_i']} LUFS -> {script.setting('loudness')} LUFS target") 

300 info(f" captions: out/{ep.name}.srt ({n} cues) chapters: out/{ep.name}.chapters.txt") 

301 if problems: 

302 info(" YouTube chapter rules (only matter when you upload; normal for a short script):") 

303 for p in problems: 

304 info(f" note: {p}") 

305 return final
