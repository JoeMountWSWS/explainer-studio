1"""Music beds: an instrumental track from the ElevenLabs Music API, sized to an episode. 

2 

3 audio/music/<hash>.mp3 cached API result (48 kHz stereo); the same prompt, length and model is never re-billed 

4 audio/music/<name>.mp3 the bed itself (or --out), ready for `music:` in script.md 

5 

6The render loops a bed shorter than the episode and trims a longer one, so the length only has to 

7be about right. Uses the same ELEVENLABS_API_KEY as narration. 

8""" 

9 

10from __future__ import annotations 

11 

12import json 

13import math 

14import shutil 

15from pathlib import Path 

16 

17from .common import ROOT, die, ffprobe_duration, info, run, sha 

18from .voice import elevenlabs_key, elevenlabs_post 

19 

20URL = "https://api.elevenlabs.io/v1/music?output_format=mp3_48000_192" 

21MODEL = "music_v2" 

22MIN_S, MAX_S = 3, 600 # API limits on music_length_ms 

23USD_PER_SECOND = 0.0083 # third-party figure; check elevenlabs.io/pricing/api 

24TAIL_S = 2 # beds run slightly past the narration so the final fade has music under it 

25 

26 

27def episode_length(ep: Path) -> float: 

28 path = ep / "timing.json" 

29 if not path.exists(): 

30 die(f"{path.relative_to(ROOT)} not found - run: studio sync {ep.name}") 

31 return float(json.loads(path.read_text(encoding="utf-8"))["duration"]) 

32 

33 

34def generate(ep: Path, prompt: str, length_s: int, model: str, variant: int = 0) -> Path: 

35 """Return the cached bed for these inputs, calling the API only on a miss. `variant` forces a fresh take.""" 

36 length_s = min(max(length_s, MIN_S), MAX_S) 

37 cache = ep / "audio" / "music" / f"{sha(prompt, length_s, model, variant)[:20]}.mp3" 

38 if cache.exists(): 

39 info(f" cached: {cache.relative_to(ROOT)}") 

40 return cache 

41 info(f" composing {length_s}s with {model} (about ${length_s * USD_PER_SECOND:.2f})") 

42 body = {"prompt": prompt, "music_length_ms": length_s * 1000, "model_id": model, "force_instrumental": True} 

43 audio = elevenlabs_post(URL, elevenlabs_key(), body, timeout=600) 

44 cache.parent.mkdir(parents=True, exist_ok=True) 

45 raw = cache.with_name(f".{cache.stem}.raw.mp3") 

46 raw.write_bytes(audio) 

47 run(["ffmpeg", "-v", "error", "-y", "-i", str(raw), "-ar", "48000", "-ac", "2", "-c:a", "libmp3lame", 

48 "-b:a", "192k", str(cache)]) 

49 raw.unlink() 

50 return cache

53def make_beds(ep: Path, prompt: str, name: str, length_s: int | None, model: str, variants: int, 

54 out: Path | None) -> list[Path]: 

55 length = length_s or math.ceil(episode_length(ep)) + TAIL_S 

56 dst_dir = out.parent if out else ep / "audio" / "music" 

57 dst_dir.mkdir(parents=True, exist_ok=True) 

58 stem = out.stem if out else name 

59 paths = [] 

60 for i in range(variants): 

61 suffix = f"-{chr(ord('a') + i)}" if variants > 1 else "" 

62 dst = dst_dir / f"{stem}{suffix}.mp3" 

63 shutil.copyfile(generate(ep, prompt, length, model, i), dst) 

64 paths.append(dst) 

65 info(f" {dst.relative_to(ROOT) if dst.is_relative_to(ROOT) else dst} ({ffprobe_duration(dst):.0f}s)") 

66 return paths
