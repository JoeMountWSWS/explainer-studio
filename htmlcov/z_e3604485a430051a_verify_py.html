18LOUDNESS_TOL = 1.0 # LU either side of the `loudness` target 

19TRUE_PEAK_TOL = 0.3 # dB above the `true_peak` ceiling 

20BLACK_MIN = 0.5 # seconds of black before it is reported 

21FREEZE_MIN = 15.0 # seconds of identical frames before it is reported (a held quote or stat is normal) 

22FADE_WINDOW = 0.6 # black this close to a scene boundary is the exit fade, not a fault 

23SILENCE_DB = -50 

24SILENCE_SLACK = 0.5 # seconds beyond the longest planned gap 

25SPEECH_WARN, SPEECH_ERROR = 0.90, 0.75 

26SPEECH_LISTED = 10

36def _ffmpeg(path: Path, *args: str) -> str: 

37 """Run an analysis pass over `path` and return ffmpeg's log (the filters report on stderr).""" 

38 proc = subprocess.run(["ffmpeg", "-hide_banner", "-nostats", "-i", str(path), *args, "-f", "null", "-"], 

39 capture_output=True, text=True, encoding="utf-8", errors="replace") 

40 if proc.returncode != 0: 40 ↛ 41line 40 didn't jump to line 41 because the condition on line 40 was never true

41 die(f"ffmpeg could not read {path.name}:\n" + "\n".join(proc.stderr.strip().splitlines()[-10:])) 

42 return proc.stderr

47def check_loudness(path: Path, target: float, ceiling: float) -> Finding: 

48 log = _ffmpeg(path, "-vn", "-af", "ebur128=peak=true") 

49 summary = log[log.rindex("Summary:"):] if "Summary:" in log else "" 

50 i = re.search(r"\bI:\s+(-?[\d.]+|-inf) LUFS", summary) 

51 tp = re.search(r"Peak:\s+(-?[\d.]+|-inf) dBFS", summary) 

52 if not i or not tp: 52 ↛ 53line 52 didn't jump to line 53 because the condition on line 52 was never true

53 return Finding("error", "loudness", "could not measure (no audio?)") 

54 loud, peak = float(i.group(1)), float(tp.group(1)) 

55 text = f"{loud:.1f} LUFS (target {target:g}), true peak {peak:.1f} dBTP (ceiling {ceiling:g})" 

56 if abs(loud - target) > LOUDNESS_TOL or peak > ceiling + TRUE_PEAK_TOL: 

57 return Finding("error", "loudness", text) 

58 return Finding("ok", "loudness", text)

71def check_video(path: Path, timing: dict) -> list[Finding]: 

72 """Black frames (error) and frozen video (warning) away from scene boundaries.""" 

73 log = _ffmpeg(path, "-an", "-vf", f"blackdetect=d={BLACK_MIN}:pix_th=0.10,freezedetect=n=-50dB:d={FREEZE_MIN}") 

74 bounds = _boundaries(timing)

76 def at_boundary(a: float, b: float) -> bool: 

77 return any(abs(a - t) <= FADE_WINDOW and abs(b - t) <= FADE_WINDOW for t in bounds) 

78 

79 black = [(float(a), float(b)) for a, b in re.findall(r"black_start:([\d.]+) black_end:([\d.]+)", log)] 

80 black = [(a, b) for a, b in black if not at_boundary(a, b)] 

81 starts = [float(x) for x in re.findall(r"freeze_start: ([\d.]+)", log)] 

82 durs = [float(x) for x in re.findall(r"freeze_duration: ([\d.]+)", log)] 

83 frozen = [(a, a + d) for a, d in zip(starts, durs)] 

84 

85 out = [] 

86 if black: 

87 spans = ", ".join(f"{_hms(a)}-{_hms(b)}" for a, b in black[:5]) 

88 out.append(Finding("error", "black frames", f"{len(black)} stretch(es) of black: {spans}")) 

89 else: 

90 out.append(Finding("ok", "black frames", "none")) 

91 if frozen: 91 ↛ 92line 91 didn't jump to line 92 because the condition on line 91 was never true

92 spans = ", ".join(f"{_hms(a)}-{_hms(b)}" for a, b in frozen[:5]) 

93 out.append(Finding("warn", "frozen video", f"{len(frozen)} stretch(es) frozen for {FREEZE_MIN:g}s+: {spans}")) 

94 else: 

95 out.append(Finding("ok", "frozen video", f"nothing frozen for {FREEZE_MIN:g}s+")) 

96 return out

101def _subtract(a: float, b: float, holes: list[tuple[float, float]]) -> list[tuple[float, float]]: 

102 pieces = [(a, b)] 

103 for x, y in holes: 

104 nxt = [] 

105 for p, q in pieces: 

106 if y <= p or x >= q: 

107 nxt.append((p, q)) 

108 continue 

109 if x > p: 

110 nxt.append((p, x)) 

111 if y < q: 

112 nxt.append((y, q)) 

113 pieces = nxt 

114 return pieces

117def silence_limit(script: Script, timing: dict) -> float: 

118 """Longest silence the script itself plans: a {pause}, a paragraph gap, or one scene's pads, plus slack.""" 

119 gaps = [c.gap_after for sc in script.scenes for c in sc.chunks] 

120 planned = max(gaps + [float(script.setting("paragraph_gap")), 

121 float(script.setting("pad_in")) + float(script.setting("pad_out"))]) 

122 return planned + SILENCE_SLACK

125def check_silence(path: Path, script: Script, timing: dict) -> Finding: 

126 limit = silence_limit(script, timing) 

127 log = _ffmpeg(path, "-vn", "-af", f"silencedetect=noise={SILENCE_DB}dB:d={limit:g}") 

128 starts = [float(x) for x in re.findall(r"silence_start: (-?[\d.]+)", log)] 

129 ends = [float(x) for x in re.findall(r"silence_end: ([\d.]+)", log)] 

130 ends += [timing["duration"]] * (len(starts) - len(ends)) # silent to the very end 

131 # Scenes without narration are allowed to be quiet (their own pads still count). 

132 quiet = [(s["start"], s["start"] + s["duration"]) for s in timing["scenes"] if s["voEnd"] <= s["voStart"]] 

133 bad = [] 

134 for a, b in zip(starts, ends): 

135 bad += [(p, q) for p, q in _subtract(max(a, 0.0), b, quiet) if q - p > limit] 

136 if bad: 

137 spans = ", ".join(f"{_hms(p)}-{_hms(q)} ({q - p:.1f}s)" for p, q in bad[:5]) 

138 return Finding("error", "silence", f"quiet for longer than {limit:.1f}s: {spans}") 

139 return Finding("ok", "silence", f"no gap longer than {limit:.1f}s")

144def check_format(path: Path, timing: dict) -> list[Finding]: 

145 proc = run(["ffprobe", "-v", "error", "-show_entries", "stream=codec_type,width,height,r_frame_rate:format=duration", 

146 "-of", "json", str(path)]) 

147 probe = json.loads(proc) 

148 video = next((s for s in probe["streams"] if s["codec_type"] == "video"), None) 

149 audio = next((s for s in probe["streams"] if s["codec_type"] == "audio"), None) 

150 out = [] 

151 if not video: 151 ↛ 152line 151 didn't jump to line 152 because the condition on line 151 was never true

152 return [Finding("error", "format", "no video stream")] 

153 num, den = (int(x) for x in video["r_frame_rate"].split("/")) 

154 fps = num / den 

155 want = f"{timing['width']}x{timing['height']} @ {timing['fps']} fps" 

156 got = f"{video['width']}x{video['height']} @ {fps:g} fps" 

157 ok = (video["width"], video["height"]) == (timing["width"], timing["height"]) and abs(fps - timing["fps"]) < 0.01 

158 out.append(Finding("ok" if ok else "error", "format", got if ok else f"{got}, expected {want}")) 

159 dur, frame = float(probe["format"]["duration"]), 1 / timing["fps"] 

160 text = f"{dur:.2f}s (timing.json says {timing['duration']:.2f}s)" 

161 out.append(Finding("ok" if abs(dur - timing["duration"]) <= 3 * frame else "error", "duration", text)) 

162 if not audio: 162 ↛ 163line 162 didn't jump to line 163 because the condition on line 162 was never true

163 out.append(Finding("error", "audio stream", "missing")) 

164 return out

173def check_speech(path: Path, script: Script) -> Finding: 

174 """Transcribe the mastered audio and compare it with the script: does the voice survive the music bed?""" 

175 want = _words(script) 

176 if not want: 

177 return Finding("ok", "speech", "no narration to check") 

178 with tempfile.TemporaryDirectory() as tmp: 

179 wav = Path(tmp) / "master.wav" 

180 run(["ffmpeg", "-v", "error", "-y", "-i", str(path), "-vn", "-ac", "1", "-ar", str(SAMPLE_RATE), str(wav)]) 

181 heard = [n for w in voice.transcribe(wav) if (n := voice._norm(w["text"]))] 

182 match = difflib.SequenceMatcher(None, want, heard, autojunk=False) 

183 hit = [False] * len(want) 

184 for blk in match.get_matching_blocks(): 

185 for k in range(blk.size): 

186 hit[blk.a + k] = True 

187 ratio = sum(hit) / len(want) 

188 text = f"{ratio:.0%} of {len(want)} script words heard" 

189 level = "error" if ratio < SPEECH_ERROR else "warn" if ratio < SPEECH_WARN else "ok" 

190 if level != "ok": 

191 missed = [w for w, h in zip(want, hit) if not h] 

192 text += "; not heard: " + " ".join(missed[:SPEECH_LISTED]) + (" ..." if len(missed) > SPEECH_LISTED else "") 

193 return Finding(level, "speech", text)

198def check_deliverables(slug: str, timing: dict) -> list[Finding]: 

199 out = [] 

200 srt, chap = OUT / f"{slug}.srt", OUT / f"{slug}.chapters.txt" 

201 out.append(Finding("ok" if srt.exists() and srt.stat().st_size else "error", "captions", f"out/{srt.name}")) 

202 if not chap.exists() or not chap.stat().st_size: 

203 return out + [Finding("error", "chapters", f"out/{chap.name} missing")] 

204 lines = chap.read_text(encoding="utf-8").splitlines() 

205 if not lines[0].startswith("0:00"): 

206 out.append(Finding("error", "chapters", "the first chapter must start at 0:00")) 

207 elif timing.get("format") == "reel": 207 ↛ 208line 207 didn't jump to line 208 because the condition on line 207 was never true

208 out.append(Finding("ok", "chapters", "reel (chapters are a YouTube long-form feature)")) 

209 else: 

210 problems = chapters(timing)[1] 

211 out.append(Finding("warn", "chapters", "; ".join(problems)) if problems 

212 else Finding("ok", "chapters", f"{len(lines)} chapters, first at 0:00, each 10s or longer")) 

213 return out

218def verify_master(ep: Path, script: Script, timing: dict, master: Path, speech: bool = True, 

219 deliverables: bool = True) -> list[Finding]: 

220 if not master.exists(): 220 ↛ 222line 220 didn't jump to line 222 because the condition on line 220 was always true

221 die(f"no master at {master}: run `studio render {ep.name}` first") 

222 findings = check_format(master, timing) 

223 findings.append(check_loudness(master, float(script.setting("loudness")), float(script.setting("true_peak"))))

233def report(findings: list[Finding]) -> int: 

234 """Print one line per finding, `setup --check` style. Returns the number of errors.""" 

235 for f in findings: 

236 info(f" {f.level.replace('error', 'ERROR'):<6} {f.name:<14} {f.message}") 

237 errors = sum(f.level == "error" for f in findings) 

238 warns = sum(f.level == "warn" for f in findings) 

239 info(f"\n {errors} error(s), {warns} warning(s)") 

240 return errors
